# «Повторить» Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Кнопка «Повторить» даёт серию из 15 уже решавшихся заданий (сначала грубые ошибки, потом опечатки «почти», потом остальное с весом по уровню), раз в день бесплатно; безлимит у «Год» и купленного «Уровня»; любая разница в одну букву в слове от 4 букв прощается с «почти».

**Architecture:** Чистая функция отбора (`internal/repeat`) отдельно от HTTP. Хранилище получает колонку `attempts.typo` и таблицу `repeat_sessions`. `GET /api/repeat` заменяет `GET /api/mistakes`, лимит и безлимит решает сервер (402). Фронтенд: `MistakesView` → `RepeatView`, окно лимита, кнопка на карте.

**Tech Stack:** Go 1.25 (stdlib `net/http`, SQLite/Postgres через `store`), Vue 3 + Pinia + Vitest, миграции `server/internal/store/migrations/NNN_*.sql`.

**Spec:** `docs/superpowers/specs/2026-10-23-repeat-design.md`

## Global Constraints

- Не коммитить, не пушить, не деплоить: владелец курса просил ничего не коммитить без его слова. Вместо «commit» в каждой задаче — прогон тестов.
- Ответы сервера никогда не содержат `answer`, `accept`, `pairs`, `say` (как у `mistakes`): только через `toExerciseDTO`.
- Серия = 15 заданий (`repeat.Size`); тиры по порядку: грубые ошибки → «почти» → остальное по весу `уровень × (1 + min(дней,30)/10)`.
- Прощение: ровно одна правка в одном слове, слово в ожидаемом ответе не короче 4 букв; `conjugate` остаётся на строгом правиле (защищены последние 2–3 буквы).
- День ученика — `userLocation` (часовой пояс из `users.timezone`), как у серии и цели дня.
- Безлимит: `AllAccessUntil` активен ИЛИ `HasPaidLevel` (оплата рублями «Уровня»; тестовый платёж считается). Уровень за зёрнышки безлимит не даёт.
- Любая точка входа в оплату проверяет `tariffs.enabled` (CLAUDE.md, «Тарифы»).
- Комментарии в коде по-английски, тексты для пользователя по-русски.
- Тесты рядом с кодом; `go test ./server/...` и `cd web && npx vitest run && npx vue-tsc -b` должны быть зелёными.

```go
// slip reports whether typed is the expected word with only a spelling slip.
// lenient is the everyday rule: one letter anywhere in a word of at least four
// letters. The strict rule (conjugation tables) still protects the ending.
func slip(typed, want string, lenient bool) bool {
	if typed == want {
		return false
	}
	if fold(typed) == fold(want) {
		return true // diacritics only
	}
	w := []rune(want)
	if lenient {
		if len(w) < 4 {
			return false
		}
		return oneEditBefore([]rune(typed), w, len(w)+1) // the limit is past every position
	}
	if len(w) < 5 {
		return false
	}
	limit := len(w) - protectedTail(len(w)) // edits must happen before this index
	return oneEditBefore([]rune(typed), w, limit)
}
```

```go
package store

import (
	"database/sql"
	"encoding/json"
	"errors"
	"fmt"
	"strings"
	"time"
)

// RepeatCandidate is an exercise the learner has answered at least once, with what
// "Повторить" needs to rank it.
type RepeatCandidate struct {
	ExerciseID  string
	Lesson      string
	LastCorrect bool
	LastTypo    bool
	Wrong       int
	Total       int
	LastAt      time.Time
}

// RepeatCandidates lists every exercise the learner has attempted, with the outcome of
// the latest attempt and lifetime wrong/total counts.
func (u *UserStore) RepeatCandidates() ([]RepeatCandidate, error) {
	rows, err := u.db.Query(`
SELECT a.exercise_id, a.lesson, a.correct, a.typo, a.attempted_at, w.wrong, w.total
FROM attempts a
JOIN (SELECT exercise_id, MAX(id) AS last_id, SUM(1 - correct) AS wrong, COUNT(*) AS total
      FROM attempts WHERE user_id = ? GROUP BY exercise_id) w ON a.id = w.last_id`, u.user)
	if err != nil {
		return nil, fmt.Errorf("repeat candidates: %w", err)
	}
	defer rows.Close()
	var out []RepeatCandidate
	for rows.Next() {
		var c RepeatCandidate
		var correct, typo int
		var at string
		if err := rows.Scan(&c.ExerciseID, &c.Lesson, &correct, &typo, &at, &c.Wrong, &c.Total); err != nil {
			return nil, fmt.Errorf("repeat candidates: %w", err)
		}
		c.LastCorrect, c.LastTypo = correct == 1, typo == 1
		c.LastAt, _ = time.Parse(time.RFC3339, at)
		out = append(out, c)
	}
	return out, rows.Err()
}

// Today is the learner's calendar date at now, in their own timezone.
func (u *UserStore) Today(now time.Time) string {
	return now.In(userLocation(u.db, u.user)).Format(dateFmt)
}

// RepeatSession is one learner's batch for one day. Keys are "lesson|exerciseID".
type RepeatSession struct {
	ID        int64
	Day       string
	Keys      []string
	StartedAt time.Time
}

// RepeatSession returns the batch stored for day, nil when there is none.
func (u *UserStore) RepeatSession(day string) (*RepeatSession, error) {
	var s RepeatSession
	var raw, started string
	err := u.db.QueryRow(`SELECT id, day, exercise_ids, started_at FROM repeat_sessions WHERE user_id = ? AND day = ?`,
		u.user, day).Scan(&s.ID, &s.Day, &raw, &started)
	if errors.Is(err, sql.ErrNoRows) {
		return nil, nil
	}
	if err != nil {
		return nil, fmt.Errorf("repeat session: %w", err)
	}
	if err := json.Unmarshal([]byte(raw), &s.Keys); err != nil {
		return nil, fmt.Errorf("repeat session: bad batch: %w", err)
	}
	s.StartedAt, _ = time.Parse(time.RFC3339, started)
	return &s, nil
}

// StartRepeatSession stores keys as the batch for day. A second call for the same day
// changes nothing and returns the batch already stored (two tabs, a double tap).
func (u *UserStore) StartRepeatSession(day string, keys []string, now time.Time) (*RepeatSession, error) {
	raw, _ := json.Marshal(keys)
	if _, err := u.db.Exec(`INSERT INTO repeat_sessions (user_id, day, exercise_ids, started_at) VALUES (?, ?, ?, ?)
		ON CONFLICT (user_id, day) DO NOTHING`, u.user, day, string(raw), now.UTC().Format(time.RFC3339)); err != nil {
		return nil, fmt.Errorf("start repeat session: %w", err)
	}
	return u.RepeatSession(day)
}

// RepeatSessionDone reports whether every exercise of the batch has a correct answer
// recorded at or after the batch started.
func (u *UserStore) RepeatSessionDone(s RepeatSession) (bool, error) {
	since := s.StartedAt.UTC().Format(time.RFC3339)
	for _, k := range s.Keys {
		_, ex, _ := strings.Cut(k, "|")
		var n int
		if err := u.db.QueryRow(`SELECT COUNT(*) FROM attempts WHERE user_id = ? AND exercise_id = ? AND correct = 1 AND attempted_at >= ?`,
			u.user, ex, since).Scan(&n); err != nil {
			return false, fmt.Errorf("repeat session done: %w", err)
		}
		if n == 0 {
			return false, nil
		}
	}
	return true, nil
}

// HasUnlimitedRepeat: a running year, or at least one level paid for in rubles.
// Levels bought with зёрнышки do not count.
func (s *Store) HasUnlimitedRepeat(userID string, now time.Time) (bool, error) {
	if _, ok, err := s.User(userID).AllAccessUntil(now); err != nil || ok {
		return ok, err
	}
	return s.HasPaidLevel(userID)
}
```

```go
const Size = 15

// Candidate is one exercise the learner has already answered.
type Candidate struct {
	Key           string // "lesson|exerciseID"
	Level         int    // 1 for the first level … 6 for the last; harder material comes back more often
	LastCorrect   bool
	LastTypo      bool // the last answer was right only because a one-letter slip was forgiven
	Wrong, Total  int
	LastAt        time.Time
	DaysSince     int  // whole days since the last attempt
	AnsweredToday bool // answered on the learner's calendar day already
}

// Select returns up to size keys: first every exercise whose last answer was wrong
// (most wrong first), then the almost-right ones (freshest first), then a weighted
// random draw from the rest. seed makes the draw repeatable.
func Select(cands []Candidate, size int, seed int64) []string {
	var wrong, typo, rest []Candidate
	for _, c := range cands {
		switch {
		case !c.LastCorrect:
			wrong = append(wrong, c)
		case c.LastTypo:
			typo = append(typo, c)
		case !c.AnsweredToday:
			rest = append(rest, c)
		}
	}
	sort.Slice(wrong, func(i, j int) bool {
		a, b := wrong[i], wrong[j]
		if a.Wrong != b.Wrong {
			return a.Wrong > b.Wrong
		}
		if a.Total != b.Total {
			return a.Total > b.Total
		}
		return a.Key < b.Key
	})
	sort.Slice(typo, func(i, j int) bool {
		a, b := typo[i], typo[j]
		if !a.LastAt.Equal(b.LastAt) {
			return a.LastAt.After(b.LastAt)
		}
		return a.Key < b.Key
	})

	var out []string
	for _, tier := range [][]Candidate{wrong, typo} {
		for _, c := range tier {
			if len(out) == size {
				return out
			}
			out = append(out, c.Key)
		}
	}
	return append(out, draw(rest, size-len(out), seed)...)
}

// draw picks n of cands without repeats, each with probability proportional to its weight
// (Efraimidis–Spirakis: the n largest u^(1/w)).
func draw(cands []Candidate, n int, seed int64) []string {
	if n <= 0 || len(cands) == 0 {
		return nil
	}
	sort.Slice(cands, func(i, j int) bool { return cands[i].Key < cands[j].Key }) // a stable base for the seed
	rng := rand.New(rand.NewSource(seed))
	type scored struct {
		key string
		s   float64
	}
	all := make([]scored, len(cands))
	for i, c := range cands {
		w := float64(max(c.Level, 1)) * (1 + float64(min(c.DaysSince, 30))/10)
		all[i] = scored{c.Key, math.Pow(rng.Float64(), 1/w)}
	}
	sort.Slice(all, func(i, j int) bool {
		if all[i].s != all[j].s {
			return all[i].s > all[j].s
		}
		return all[i].key < all[j].key
	})
	out := make([]string, 0, n)
	for i := 0; i < len(all) && i < n; i++ {
		out = append(out, all[i].key)
	}
	return out
}
```
